app.get('/api/sessions/:id/order', requireUser, validateSessionOwnership, async (req, res) => {
  try {
    const { id } = req.params
    const session = req.session
    
    if (!session.order_no) {
      return res.json({ success: true, data: null, message: '该会话未绑定订单' })
    }
    
    // 如果有缓存的订单数据直接返回，否则从订单系统获取
    if (session.order_data) {
      return res.json({ success: true, data: session.order_data, cached: true })
    }
    
    // 从订单系统获取最新订单数据
    const url = `${ORDER_SYSTEM_URL}/api/external/order/${session.order_no}`
    const response = await fetch(url, {
      headers: {
        'X-Api-Key': APPEAL_API_KEY,
        'Content-Type': 'application/json'
      }
    })
    
    if (!response.ok) {
      return res.status(response.status).json({ error: '获取订单详情失败' })
    }
    
    const result = await response.json()
    if (result.success && result.data) {
      // 更新缓存
      await updateSessionOrder(id, session.order_no, result.data)
      return res.json({ success: true, data: result.data, cached: false })
    }
    
    res.json({ success: true, data: null })
  } catch (error) {
    console.error('获取会话订单失败:', error)
    res.status(500).json({ error: '获取订单信息失败' })
  }
})

app.put('/api/sessions/:id/field', requireUser, validateSessionOwnership, async (req, res) => {
  try {
    const session = req.session
    const { key, value } = req.body
    if (!key || key.length > 50) return res.status(400).json({ error: '字段名无效' })
    if (value && value.length > 2000) return res.status(400).json({ error: '字段内容过长' })
    const collectedData = session.collected_data || {}
    const oldValue = collectedData[key] || ''
    const normalizedValue = normalizeFieldValue(key, value || '', collectedData)
    if (normalizedValue !== oldValue) {
      const fieldDef = INFO_FIELDS.find(f => f.key === key)
      const label = fieldDef?.label || key
      const reason = oldValue && oldValue !== '用户暂未提供' && oldValue !== '⏳待补充'
        ? `用户手动将"${label}"从"${oldValue}"修改为"${normalizedValue}"`
        : `用户手动填写了"${label}"`
      logFieldChange(req.params.id, key, label, oldValue, normalizedValue, 'user_edit', reason).catch(() => {})
    }
    collectedData[key] = normalizedValue
    await updateSession(req.params.id, session.step, collectedData)
    res.json({
      success: true,
      collectedData,
      normalizedValue,
      wasNormalized: normalizedValue !== (value || '').trim(),
    })
  } catch (err) {
    console.error(err)
    res.status(500).json({ error: '更新失败' })
  }
})
